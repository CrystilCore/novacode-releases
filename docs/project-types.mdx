<AccordionGroup>
  <Accordion title="Scratch">
    Compiles to a plain `.sb3` that runs in Scratch itself.
  </Accordion>
  <Accordion title="TurboWarp">
    For TurboWarp: its extensions load and work.
  </Accordion>
  <Accordion title="Nova">
    NovaCode's own player: real types (arrays and dicts), background jobs, and export as a web folder or an app.
  </Accordion>
  <Accordion title="Code">
    JavaScript files that run in the NovaCode Runner window.
  </Accordion>
</AccordionGroup>
